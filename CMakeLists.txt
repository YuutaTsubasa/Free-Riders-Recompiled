# files runs the C compiler as rc's preprocessor and fails with clang-cl; the
# resource includes nothing, so there are no dependencies to scan.
set(CMAKE_NINJA_CMCLDEPS_RC OFF)
set(CMAKE_CXX_STANDARD 20)

# libc++ (the NDK's, and older LLVM's on macOS) ships std::jthread and
# std::stop_token as experimental. Newer Apple libc++ has them as standard,
# and the iOS Simulator SDK has no experimental library at all: on Apple
# platforms the flag is used only where std::jthread needs it.
set(SFR_EXPERIMENTAL_LIBRARY ${ANDROID})
if(APPLE)
    include(CheckCXXSourceCompiles)
    set(CMAKE_REQUIRED_FLAGS "-std=c++20")
    check_cxx_source_compiles("#include <thread>
int main() { std::jthread t([](std::stop_token) {}); }" SFR_STANDARD_JTHREAD)
    unset(CMAKE_REQUIRED_FLAGS)
    if(NOT SFR_STANDARD_JTHREAD)
        set(SFR_EXPERIMENTAL_LIBRARY ON)
    endif()
endif()
if(SFR_EXPERIMENTAL_LIBRARY)
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fexperimental-library>)
    add_link_options(-fexperimental-library)
endif()
if(APPLE)
    # The pinned fmt uses malloc without including <cstdlib>, which newer
    # libc++ no longer brings in on its own.
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:SHELL:-include cstdlib>")
endif()

# Profile-guided optimisation with Clang (docs/performance.md): "generate"
# builds an instrumented game that writes default_*.profraw into its working
# directory when it exits (on Android the app's files directory); a merged
# .profdata file (llvm-profdata merge) builds the game optimised with it.
# Functions the profile does not match are compiled as without it.
set(SFR_PGO "" CACHE STRING "Profile-guided optimisation: empty (off), generate, or a .profdata file")
if(SFR_PGO)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
        message(FATAL_ERROR "SFR_PGO needs Clang with its GNU-style driver (Android, Linux, macOS).")
    endif()
    if(SFR_PGO STREQUAL "generate")
        add_compile_options(-fprofile-generate)
        add_link_options(-fprofile-generate)
        add_compile_definitions(SFR_PGO_GENERATE=1)
    else()
        get_filename_component(SFR_PGO_PROFILE "${SFR_PGO}" ABSOLUTE)
        if(NOT EXISTS "${SFR_PGO_PROFILE}")
            message(FATAL_ERROR "SFR_PGO: no profile at ${SFR_PGO_PROFILE}")
        endif()
        # Ninja does not see the profile as an input: a new profile at the
        # same path needs a clean build.
        add_compile_options("-fprofile-use=${SFR_PGO_PROFILE}" -Wno-profile-instr-unprofiled -Wno-profile-instr-out-of-date -Wno-profile-instr-missing)
    endif()
endif()

    # Window, input and sound on Linux and Android.
    if(ANDROID)
        # Android has no system SDL2: SDL 2.32.8 from source (tools/SDL),
        # shared, because its Java activity loads libSDL2.so, then libmain.so.
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/SDL/CMakeLists.txt")
            message(FATAL_ERROR "Run scripts/bootstrap.py to obtain the pinned SDL.")
        endif()
        set(SDL_SHARED ON CACHE BOOL "" FORCE)
        set(SDL_STATIC OFF CACHE BOOL "" FORCE)
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "" FORCE)
        add_subdirectory(tools/SDL EXCLUDE_FROM_ALL)
        set(SDL2_VERSION 2.32.8)
    elseif(APPLE)
        # macOS: the same pinned SDL, linked in, so FreeRidersRecompiled.app
        # needs no Homebrew and runs on the macOS set above.
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/SDL/CMakeLists.txt")
            message(FATAL_ERROR "Run scripts/bootstrap.py to obtain the pinned SDL.")
        endif()
        enable_language(OBJC)  # SDL's Cocoa code; a language enabled only inside its directory fails to generate
        if(IOS)
            # iOS: the launcher and the game share one process (the game is
            # a library the launcher loads, as on Android), so one shared SDL;
            # SDL2main is UIKit's entry, which calls the launcher's SDL_main.
            enable_language(OBJCXX)
            set(SDL_SHARED ON CACHE BOOL "" FORCE)
            set(SDL_STATIC OFF CACHE BOOL "" FORCE)
            set(SDL2_DISABLE_SDL2MAIN OFF CACHE BOOL "" FORCE)
            add_compile_definitions(SFR_IOS=1)
        else()
            set(SDL_SHARED OFF CACHE BOOL "" FORCE)
            set(SDL_STATIC ON CACHE BOOL "" FORCE)
            set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "" FORCE)
        endif()
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        add_subdirectory(tools/SDL EXCLUDE_FROM_ALL)
        if(NOT TARGET SDL2::SDL2)
            add_library(SDL2::SDL2 ALIAS SDL2-static)
        endif()
        set(SDL2_VERSION 2.32.8)
    else()
        find_package(SDL2 REQUIRED)
    endif()
endif()

function(sfr_copy_onnx_dlls target)
    if(TARGET sfr_onnx_dlls)
        add_dependencies(${target} sfr_onnx_dlls)
    endif()
endfunction()
# Finding a body in a camera picture: ONNX Runtime and the pinned MediaPipe
# models (tools/onnx, fetched by scripts/fetch_pose_model.py). Without them
# the camera can still give the title a picture, but not drive the player.
# Android takes the runtime fetched for each ABI (--android).
if(ANDROID)
    set(SFR_ONNX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/tools/onnx/onnxruntime-android")
    set(SFR_ONNX_LIBRARY "${SFR_ONNX_ROOT}/lib/${ANDROID_ABI}/libonnxruntime.so")
else()
    set(SFR_ONNX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/tools/onnx/onnxruntime" CACHE PATH "Host ONNX Runtime installation")
    if(WIN32)
        set(SFR_ONNX_LIBRARY "${SFR_ONNX_ROOT}/lib/onnxruntime.lib")
    else()
        set(SFR_ONNX_LIBRARY "${SFR_ONNX_ROOT}/lib/libonnxruntime.so")
    endif()
endif()
if(EXISTS "${SFR_ONNX_ROOT}/include/onnxruntime_cxx_api.h" AND EXISTS "${SFR_ONNX_LIBRARY}")
    set(SFR_POSE_AVAILABLE ON)
    add_library(sfr_onnxruntime INTERFACE)
    target_include_directories(sfr_onnxruntime INTERFACE "${SFR_ONNX_ROOT}/include")
    target_link_libraries(sfr_onnxruntime INTERFACE "${SFR_ONNX_LIBRARY}")
    if(NOT WIN32 AND NOT ANDROID)
        # A release keeps libonnxruntime.so.1 beside the program.
        target_link_options(sfr_onnxruntime INTERFACE "LINKER:-rpath,$ORIGIN")
    endif()
    # Windows finds a DLL beside the program before System32, which has an
    # older onnxruntime.dll of its own (Windows ML's): without the fetched
    # one beside it, the program loads that one and fails to start.
    if(WIN32)
        file(GLOB SFR_ONNX_DLLS "${SFR_ONNX_ROOT}/lib/*.dll")
        set(SFR_ONNX_COPIES "")
        foreach(dll ${SFR_ONNX_DLLS})
            get_filename_component(name "${dll}" NAME)
            list(APPEND SFR_ONNX_COPIES "${CMAKE_BINARY_DIR}/${name}")
            add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/${name}"
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "${CMAKE_BINARY_DIR}/${name}" DEPENDS "${dll}")
        endforeach()
        if(SFR_ONNX_COPIES)
            add_custom_target(sfr_onnx_dlls DEPENDS ${SFR_ONNX_COPIES})
        endif()
    endif()
    set(SFR_POSE_SOURCES src/pose_estimator.cpp src/pose_estimator_onnx.cpp src/pose_mediapipe.cpp src/pose_mediapipe_onnx.cpp)
else()
    set(SFR_POSE_AVAILABLE OFF)
    set(SFR_POSE_SOURCES src/pose_estimator.cpp src/pose_estimator_none.cpp)
endif()

    function(sfr_embed_shader source entry profile name)
        set(header "${SFR_EMBEDDED_SHADERS}/${name}_dxil.h")
        add_custom_command(OUTPUT "${header}"
            COMMAND ${SFR_DXC} -T ${profile} -E ${entry} -HV 2021 -Qstrip_debug -Qstrip_reflect -Fh "${header}" -Vn ${name}_dxil "${CMAKE_CURRENT_SOURCE_DIR}/${source}"
            DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}" ${SFR_SHADER_INCLUDES} VERBATIM)
        set(spirv_header "${SFR_EMBEDDED_SHADERS}/${name}_spirv.h")
        set(invert)
        if(profile MATCHES "^vs_")

            DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}" ${SFR_SHADER_INCLUDES} VERBATIM)
        set(SFR_EMBEDDED_SHADER_HEADERS ${SFR_EMBEDDED_SHADER_HEADERS} "${header}" "${spirv_header}" PARENT_SCOPE)
    endfunction()

# D3D12 through the Agility SDK's runtime (scripts/fetch_d3d12_agility.py,
# run by bootstrap on Windows): every program that draws with D3D12 exports
# the SDK version it was fetched at and finds D3D12Core.dll in D3D12\ beside
# it. Without the runtime, Windows 10 cannot make Plume's command lists and
# the game falls back to Vulkan (native_graphics.cpp).
if(WIN32)
    set(SFR_D3D12_AGILITY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tools/d3d12-agility")
    if(EXISTS "${SFR_D3D12_AGILITY_DIR}/D3D12Core.dll" AND EXISTS "${SFR_D3D12_AGILITY_DIR}/version.txt")
        file(STRINGS "${SFR_D3D12_AGILITY_DIR}/version.txt" SFR_D3D12_SDK_VERSION LIMIT_COUNT 1)
        message(STATUS "D3D12 Agility SDK runtime: ${SFR_D3D12_SDK_VERSION}")
        add_library(sfr_d3d12_agility OBJECT src/d3d12_agility.cpp)
        target_compile_definitions(sfr_d3d12_agility PRIVATE SFR_D3D12_SDK_VERSION=${SFR_D3D12_SDK_VERSION})
        # One copy for all of them: they are all built into this directory.
        set(SFR_D3D12_CORE "${CMAKE_BINARY_DIR}/D3D12/D3D12Core.dll")
        add_custom_command(OUTPUT "${SFR_D3D12_CORE}"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/D3D12"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SFR_D3D12_AGILITY_DIR}/D3D12Core.dll" "${SFR_D3D12_CORE}"
            DEPENDS "${SFR_D3D12_AGILITY_DIR}/D3D12Core.dll")
        add_custom_target(sfr_d3d12_runtime DEPENDS "${SFR_D3D12_CORE}")
        foreach(target sfr_cpu_diagnostic sfr_native_graphics_test sfr_native_upload_batch_test sfr_native_shaders_test sfr_native_presentation_test
                       sfr_native_resolution_test sfr_native_pipeline_prewarm_test sfr_shader_cache_rebuild sfr_native_shader_abi_test
                       sfr_native_raster_state_test sfr_native_blend_control_test sfr_guest_graphics_test
                       sfr_guest_render_state_test sfr_guest_blend_request_test sfr_guest_sampler_filter_test
                       sfr_guest_texture_binding_test)
            if(TARGET ${target})
                target_sources(${target} PRIVATE $<TARGET_OBJECTS:sfr_d3d12_agility>)
                add_dependencies(${target} sfr_d3d12_runtime)
            endif()
        endforeach()
    else()
        message(STATUS "D3D12 Agility SDK runtime: not fetched (Windows 10 will draw with Vulkan)")
    endif()
endif()
